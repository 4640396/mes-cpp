71char* SPHashPassword( const char* plaintextPassword )
72{
73 if ( plaintextPassword == NULL || plaintextPassword[0] == '\0' )
74 {
75 throw SPHashPasswordIllegalPasswordException( "Password should not be blank." );
76 }
77 std::string encryptedPassword;
78 const char* salt = getenv( SP_ENCRYPT_PASSWORD_SALT );
79 if ( !salt || salt[0] == '\0' )
80 {
81 salt = "%2u";
82 }
83 for ( const char* p = salt; *p; p++ )
84 {
85 if ( *p != '%' )
86 {
87 encryptedPassword += *p;
88 continue;
89 }
90 ++p;
91 if ( *p == '%' )
92 {
93 encryptedPassword += *p;
94 continue;
95 }
96 else if ( !isdigit( *p ) )
97 {
98 throw SPHashPasswordIllegalSaltException( "Length expected: ", p );
99 }
100 long len = strtol( p, const_cast<char**>(&p), 10 );
101 switch ( *p )
102 {
103 case 'u':
104 case 's':
105 {
106 timeval tv;
107 gettimeofday( &tv, NULL );
108 unsigned long val = tv.tv_sec;
109 if ( *p == 'u' )
110 {
111 val = val * 1000000 + tv.tv_usec;
112 }
113 for ( int i = 0; i < len; i++ )
114 {
115 encryptedPassword += CRYPT_CHARS[val % 64];
116 val /= 64;
117 }
118 break;
119 }
120 case 'r':
121 {
122 std::ifstream ifs( RANDOM_DEVICE );
123 for ( int i = 0; i < len; i++ )
124 {
125 encryptedPassword += CRYPT_CHARS[ifs.get() % 64];
126 }
127 break;
128 }
129 default:
130 {
131 throw SPHashPasswordIllegalSaltException( "Invalid specifier: ", p );
132 }
133 }
134 }
135#ifdef _AIX
136 pthread_mutex_lock( &crypt_mutex );
137 const char* password = crypt( plaintextPassword, encryptedPassword.c_str() );
138#else
139 crypt_data data;
140 const char* password = crypt_r( plaintextPassword, encryptedPassword.c_str(), &data );
141#endif
142 char* retVal = new char[strlen( password ) + 1];
143 strcpy( retVal, password );
144#ifdef _AIX
145 pthread_mutex_unlock( &crypt_mutex );
146#endif
147 return retVal;
148}
