27class speb_Table
28{
29 int count;
30 char *markers[speb_table_max];
31 void *ptrs[speb_table_max];
32public:
33 speb_Table()
34 {
35 count = 0;
36 memset( markers, 0x00, sizeof(markers) );
37 memset( ptrs, 0x00, sizeof(ptrs) );
38 }
39
40 void * get( const char *marker )
41 {
42 for( int i = 0; i < count; i++ )
43 {
44 if( strcmp( markers[i], marker ) == 0 )
45 {
46 return ptrs[i];
47 }
48 }
49 return NULL;
50 }
51 void put( const char *marker, void * ptr )
52 {
53 markers[count] = strdup(marker);
54 ptrs[count] = ptr;
55 count++;
56 }
57};
