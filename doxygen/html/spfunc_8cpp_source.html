92//
93// Description:
94// Check the string value is numeric or not.
95//
96static CORBA::Boolean IsValueLong_digitCheck(char* value){
97/*
98 * Remove trailing space chars.
99 */
100 if ( striptail(value) == FALSE ) return FALSE;
101
102//D9000001 int len= strlen( value);
103 int len= CIMFWStrLen( value );//D9000001
104/*
105 * skip leading blan chars
106 */
107 int idx= 0;
108 while( value[idx] == ' ' && idx < len && value[idx] != 0 ){ idx++;}
109
110/*
111 * Check for +/- chars
112 */
113 if ( value[idx] == '+' || value[idx] == '-' ) idx++;
114
115/*
116 * check for integer chars
117 */
118 while( isdigit(value[idx]) && idx < len && value[idx] != 0 ){ idx++;}
119
120 if ( idx == len ) return TRUE;
121
122 return FALSE;
123}
